import streamlit as st

st.set_page_config(page_title="AI Tele-Diagnostic Assistant", page_icon="🩺", layout="centered")

st.markdown("<h1 style='text-align: center; color: #1E88E5;'>🩺 AI Tele-Diagnostic Assistant</h1>", unsafe_allow_html=True)
st.markdown("<p style='text-align: center; color: #555;'>Rural Health Center Decision Support System</p>", unsafe_allow_html=True)
st.divider()

st.subheader("📋 Patient Profile & Vitals Data")
col1, col2 = st.columns(2)
with col1:
    age = st.number_input("Patient Umar (Age in Years):", min_value=1, max_value=120, value=25)
    gender = st.selectbox("Jins (Gender):", ["Male", "Female", "Other"])
with col2:
    has_temp = st.checkbox("Bukhar Check Kia Hai?")
    temp = st.slider("Temperature (°F):", min_value=95.0, max_value=108.0, value=98.6, step=0.1) if has_temp else 0.0

col3, col4 = st.columns(2)
with col3:
    has_bp = st.checkbox("BP Check Kia Hai?")
    sys_bp = st.number_input("Systolic BP:", min_value=50, max_value=240, value=120) if has_bp else 0
with col4:
    has_pulse = st.checkbox("Pulse Check Ki Hai?")
    pulse = st.number_input("Pulse Rate (BPM):", min_value=30, max_value=220, value=72) if has_pulse else 0

st.divider()
st.subheader("🗣️ Patient Alamaat (Symptoms)")
symptoms = st.text_area("Alamaat Yahan Likhien (Urdu / Roman Urdu / English):", placeholder="e.g. dil ma dard, bukhar, kankani...")

def analyze_symptoms(age, gender, symptoms, temp, sys_bp, pulse):
    s = symptoms.lower()
    triage, disease, rec, alert_type = "GREEN (MILD)", "Normal Seasonal Flu", "Dispenser basic OTC dawai aur rest ka keh sakta hai.", "success"

    if ((temp > 103.0) or (sys_bp > 160 and sys_bp != 0) or (sys_bp < 90 and sys_bp != 0) or (pulse > 140 and pulse != 0) or
        "chest pain" in s or "seene me dard" in s or "dil ma dard" in s or "dil me dard" in s or "saans" in s or "behoshi" in s):
        triage, disease, rec, alert_type = "RED (CRITICAL)", "Severe Systemic Risk / Cardiopulmonary Crisis", "🚨 EMERGENCY! Patient ko Qareebi Bade Hospital Refer Karen.", "error"
    elif ("aankhon ke peeche" in s or "jism dard" in s or "body pain" in s or "rash" in s):
        triage, disease, rec, alert_type = "YELLOW (MODERATE)", "Dengue Fever Suspicion", "⚠️ Platelets (CBC Test) karwayen.", "warning"
    elif ("kankani" in s or "sardi" in s or "chills" in s):
        triage, disease, rec, alert_type = "YELLOW (MODERATE)", "Malaria Suspicion", "⚠️ Malaria MP Slide / ICT Strip test karwayen.", "warning"
    elif ((temp > 100.5) or "ulti" in s or "vomit" in s or "dast" in s or "motion" in s):
        triage, disease, rec, alert_type = "YELLOW (MODERATE)", "Acute Infection / Gastrointestinal Crisis", "⚠️ ORS aur hydration provide karen.", "warning"

    return triage, disease, rec, alert_type

if st.button("📊 DIAGNOSE & TRIAGE PATIENT"):
    if not symptoms.strip() and temp == 0 and sys_bp == 0 and pulse == 0:
        st.error("Please Kam se kam Alamaat (Symptoms) ya Vitals zaroor dakhil karen!")
    else:
        triage, disease, rec, alert_type = analyze_symptoms(age, gender, symptoms, temp, sys_bp, pulse)
        st.divider()
        st.subheader("📑 Diagnostic & Triage Report")
        if alert_type == "error": st.error(f"### TRIAGE STATUS: {triage}")
        elif alert_type == "warning": st.warning(f"### TRIAGE STATUS: {triage}")
        else: st.success(f"### TRIAGE STATUS: {triage}")
        
        st.info(f"**Patient:** {age} Yrs | {gender}\n\n**Suspected Condition:** {disease}\n\n**Recommendation:** {rec}")
