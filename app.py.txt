st.set_page_config(
    page_title="AI Tele-Diagnostic Assistant",
    page_icon="🩺",
    layout="centered"
)

# Custom Styling / CSS
st.markdown("""
    <style>

        color: #1E88E5;
        font-weight: bold;
    }
    .sub-title {
        text-align: center;

    }
    .stButton>button {
        width: 100%;
        background-color: #1E88E5;
        color: white;
        font-size: 18px;
        border-radius: 8px;
        height: 50px;
    }
    </style>
""", unsafe_allow_html=True)

# Title Header
st.markdown("<h1 class='main-title'>🩺 AI Tele-Diagnostic Assistant</h1>", unsafe_allow_html=True)
st.markdown("<p class='sub-title'>Rural Health Center & BHU Dispenser Decision Support System</p>", unsafe_allow_html=True)
st.divider()

# --- PATIENT FORM ---
st.subheader("📋 Patient Profile & Vitals Data")

col1, col2 = st.columns(2)

with col1:
    age = st.number_input("Patient Umar (Age in Years):", min_value=1, max_value=120, value=25)
    gender = st.selectbox("Jins (Gender):", ["Male", "Female", "Other"])

with col2:
    has_temp = st.checkbox("Bukhar Check Kia Hai?")
    temp = st.slider("Temperature (°F):", min_value=95.0, max_value=108.0, value=98.6, step=0.1) if has_temp else 0.0

col3, col4 = st.columns(2)

with col3:
    has_bp = st.checkbox("BP Check Kia Hai?")
    sys_bp = st.number_input("Systolic BP (Upar Wala BP):", min_value=50, max_value=240, value=120) if has_bp else 0

with col4:
    has_pulse = st.checkbox("Pulse Check Ki Hai?")
    pulse = st.number_input("Pulse Rate (Dil Ki Dhadkan BPM):", min_value=30, max_value=220, value=72) if has_pulse else 0

st.divider()

# --- SYMPTOMS INPUT ---
st.subheader("🗣️ Patient Alamaat (Symptoms)")
symptoms = st.text_area(
    "Alamaat Yahan Likhien (Urdu / Roman Urdu / English):",
    placeholder="e.g. dil ma dard, bukhar, kankani, sardi, 3 weeks se motion..."
)

# --- TRIAGE LOGIC ENGINE ---
def analyze_symptoms(age, gender, symptoms, temp, sys_bp, pulse):
    s = symptoms.lower()
    
    # Defaults (MILD)
    triage = "GREEN (MILD)"
    suspected_disease = "Normal Seasonal Flu / Mild Infection"
    recommendation = "Dispenser basic OTC dawai aur rest ka keh sakta hai."
    alert_type = "success"

    # 1. CRITICAL CHECKS (RED ALERT)
    if ((temp > 103.0) or (sys_bp > 160 and sys_bp != 0) or (sys_bp < 90 and sys_bp != 0) or (pulse > 140 and pulse != 0) or
        "chest pain" in s or "seene me dard" in s or "dil ma dard" in s or "dil me dard" in s or "dil dard" in s or
        "saans" in s or "behoshi" in s):
        
        triage = "RED (CRITICAL)"
        suspected_disease = "Severe Systemic Risk / Cardiopulmonary Crisis"
        recommendation = "🚨 EMERGENCY! Patient ko Qareebi Bade Hospital ya ICU Refer Karen."
        alert_type = "error"

    # 2. DENGUE / MALARIA CHECKS (YELLOW ALERT)
    elif ("aankhon ke peeche" in s or "jism dard" in s or "body pain" in s or "rash" in s or "surkh dhabbe" in s):
        triage = "YELLOW (MODERATE)"
        suspected_disease = "Dengue Fever Suspicion"
        recommendation = "⚠️ Platelets (CBC Test) karwayen aur City Doctor se Tele-consultation karen."
        alert_type = "warning"

    elif ("kankani" in s or "sardi" in s or "chills" in s or "chakar" in s or "pasina" in s):
        triage = "YELLOW (MODERATE)"
        suspected_disease = "Malaria Suspicion"
        recommendation = "⚠️ Malaria MP Slide / ICT Strip test karwayen."
        alert_type = "warning"

    # 3. GASTRO & CHRONIC CHECKS (YELLOW ALERT)
    elif ((temp > 100.5) or "ulti" in s or "vomit" in s or "vomitting" in s or "dast" in s or 
          "motion" in s or "sar dard" in s or "headache" in s):
        triage = "YELLOW (MODERATE)"
        suspected_disease = "Acute Infection / Gastrointestinal Crisis / Severe Headache"
        recommendation = "⚠️ ORS, hydration aur physician se consultation karwayen."
        alert_type = "warning"

    elif ("week" in s or "hafte" in s or "mahine" in s):
        triage = "YELLOW (MODERATE)"
        suspected_disease = "Chronic Underlying Condition"
        recommendation = "⚠️ Bimari lambe arse se hai, detailed medical tests ke liye refer karen."
        alert_type = "warning"

    return triage, suspected_disease, recommendation, alert_type

# --- ACTION BUTTON ---
if st.button("📊 DIAGNOSE & TRIAGE PATIENT"):
    if not symptoms.strip() and temp == 0 and sys_bp == 0 and pulse == 0:
        st.error("Please Kam se kam Alamaat (Symptoms) ya Vitals zaroor dakhil karen!")
    else:
        triage, disease, rec, alert_type = analyze_symptoms(age, gender, symptoms, temp, sys_bp, pulse)
        
        st.divider()
        st.subheader("📑 Diagnostic & Triage Report")

        # Display Triage Badge
        if alert_type == "error":
            st.error(f"### TRIAGE STATUS: {triage}")
        elif alert_type == "warning":
            st.warning(f"### TRIAGE STATUS: {triage}")
        else:
            st.success(f"### TRIAGE STATUS: {triage}")

        # Details Table Card
        st.info(f"""
        **Patient Profile:** {age} Years | {gender}  
        **Vitals Recorded:** Temp: {f'{temp}°F' if temp > 0 else 'Not Recorded'} | BP: {f'{sys_bp} mmHg' if sys_bp > 0 else 'Not Recorded'} | Pulse: {f'{pulse} BPM' if pulse > 0 else 'Not Recorded'}  
        
        ---
        **Suspected Condition:** {disease}  
        **Recommended Action:** {rec}
        """)
